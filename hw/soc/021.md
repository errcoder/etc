네. 이번 테스트는 **현재 `InstructionRom`, `SimpleDataRam`, `SimpleGpio`, `DBusDecoder`, `VexRiscvTop`, Simulation을 모두 맞물려 실행할 수 있도록** 전체 코드를 제시하겠습니다.

다만 한 가지 정정할 부분이 있습니다. 앞서 제시한 GPIO 테스트 프로그램의 `LUI + ADDI`는 **이번 테스트 목적에는 굳이 필요하지 않습니다.** `LUI x4, 0x10000`으로 GPIO 주소를 만들고 `ADDI x5, x0, 0x55`로 데이터를 만든 뒤 `SW x5, 0(x4)`를 실행하면 됩니다.

아래 코드는 **현재 프로젝트의 SpinalHDL 1.13.0 / 해당 VexRiscv commit** 기준입니다.

## 1. `SimpleGpio.scala`

```scala
package myvexsoc

import spinal.core._
import vexriscv.plugin._

class SimpleGpio extends Component {

  val io = new Bundle {
    val bus = slave(DBusSimpleBus())

    val value = out Bits(32 bits)
  }

  val gpioValue = Reg(Bits(32 bits)) init (0)

  // GPIO는 모든 command를 받을 수 있다.
  io.bus.cmd.ready := True

  // Store만 처리한다.
  when(io.bus.cmd.fire && io.bus.cmd.wr) {
    gpioValue := io.bus.cmd.payload.data
  }

  io.value := gpioValue

  // 현재 GPIO는 write-only peripheral
  io.bus.rsp.ready := False
  io.bus.rsp.error := False
  io.bus.rsp.data := B(0, 32 bits)
}
```

# 2. `DBusDecoder.scala`

```scala
package myvexsoc

import spinal.core._
import vexriscv.plugin._

class DBusDecoder extends Component {

  val io = new Bundle {
    val cpu = slave(DBusSimpleBus())

    val ram = master(DBusSimpleBus())
    val gpio = master(DBusSimpleBus())
  }

  // ------------------------------------------------------------
  // Address decode
  // ------------------------------------------------------------

  // Data RAM:
  // 0x0000_0000 ~ 0x0000_03FF
  val ramHit =
    io.cpu.cmd.address < U(0x00000400L, 32 bits)

  // GPIO:
  // 0x1000_0000
  val gpioHit =
    io.cpu.cmd.address === U(0x10000000L, 32 bits)

  // ------------------------------------------------------------
  // CPU -> RAM
  // ------------------------------------------------------------

  io.ram.cmd.valid := io.cpu.cmd.valid && ramHit
  io.ram.cmd.payload := io.cpu.cmd.payload

  // ------------------------------------------------------------
  // CPU -> GPIO
  // ------------------------------------------------------------

  io.gpio.cmd.valid := io.cpu.cmd.valid && gpioHit
  io.gpio.cmd.payload := io.cpu.cmd.payload

  // ------------------------------------------------------------
  // RAM / GPIO -> CPU : command ready
  // ------------------------------------------------------------

  io.cpu.cmd.ready := Mux(
    ramHit,
    io.ram.cmd.ready,
    Mux(
      gpioHit,
      io.gpio.cmd.ready,
      False
    )
  )

  // ------------------------------------------------------------
  // Remember RAM Load
  // ------------------------------------------------------------

  val ramLoadPending = Reg(Bool()) init (False)

  when(io.cpu.cmd.fire) {
    ramLoadPending := ramHit && !io.cpu.cmd.wr
  }

  when(io.cpu.rsp.ready) {
    ramLoadPending := False
  }

  // ------------------------------------------------------------
  // RAM response -> CPU
  // ------------------------------------------------------------

  io.cpu.rsp.ready := ramLoadPending && io.ram.rsp.ready
  io.cpu.rsp.data := io.ram.rsp.data
  io.cpu.rsp.error := io.ram.rsp.error
}
```

# 3. `InstructionRom.scala`

이번에는 GPIO 테스트 프로그램을 넣습니다.

```scala
package myvexsoc

import spinal.core._
import spinal.lib._
import vexriscv.plugin._

class InstructionRom extends Component {

  val io = new Bundle {
    val cmd = slave(Stream(IBusSimpleCmd()))
    val rsp = master(Flow(IBusSimpleRsp()))
  }

  // ------------------------------------------------------------
  // Instructions
  // ------------------------------------------------------------

  // ADDI x1, x0, 1
  val ADDI_X1_X0_1 = B"x00100093"

  // SW x1, 0(x0)
  val SW_X1_0_X0 = B"x00102023"

  // ADDI x2, x0, 2
  val ADDI_X2_X0_2 = B"x00200113"

  // SW x2, 4(x0)
  val SW_X2_4_X0 = B"x00202223"

  // LUI x4, 0x10000
  // x4 = 0x10000000
  val LUI_X4_GPIO = B"x10000237"

  // ADDI x5, x0, 0x55
  val ADDI_X5_X0_55 = B"x05500293"

  // SW x5, 0(x4)
  // GPIO[0] = 0x55
  val SW_X5_0_X4 = B"x00522023"

  // LW x3, 0(x0)
  val LW_X3_0_X0 = B"x00002183"

  // SW x3, 8(x0)
  val SW_X3_8_X0 = B"x00302423"

  // JAL x0, 0
  val JAL_X0_0 = B"x0000006F"

  val NOP = B"x00000013"

  // ------------------------------------------------------------
  // Instruction ROM
  // ------------------------------------------------------------

  io.cmd.ready := True

  val instruction = Bits(32 bits)
  instruction := NOP

  switch(io.cmd.pc) {

    is(U"32'h80000000") {
      instruction := ADDI_X1_X0_1
    }

    is(U"32'h80000004") {
      instruction := SW_X1_0_X0
    }

    is(U"32'h80000008") {
      instruction := ADDI_X2_X0_2
    }

    is(U"32'h8000000C") {
      instruction := SW_X2_4_X0
    }

    is(U"32'h80000010") {
      instruction := LUI_X4_GPIO
    }

    is(U"32'h80000014") {
      instruction := ADDI_X5_X0_55
    }

    is(U"32'h80000018") {
      instruction := SW_X5_0_X4
    }

    is(U"32'h8000001C") {
      instruction := LW_X3_0_X0
    }

    is(U"32'h80000020") {
      instruction := SW_X3_8_X0
    }

    is(U"32'h80000024") {
      instruction := JAL_X0_0
    }
  }

  // ------------------------------------------------------------
  // IBus response
  // ------------------------------------------------------------

  io.rsp.valid := RegNext(io.cmd.fire) init (False)

  io.rsp.inst := RegNextWhen(
    instruction,
    io.cmd.fire
  ) init (NOP)

  io.rsp.error := False
}
```

```text

0x80000004  SW   x1, 0(x0)          RAM[0] = 1

0x80000008  ADDI x2, x0, 2
0x8000000C  SW   x2, 4(x0)          RAM[1] = 2

0x80000010  LUI  x4, 0x10000        x4 = 0x10000000
0x80000014  ADDI x5, x0, 0x55       x5 = 0x55
0x80000018  SW   x5, 0(x4)           GPIO = 0x55

0x8000001C  LW   x3, 0(x0)           x3 = RAM[0] = 1
```

# 4. `SimpleDataRam.scala`

기존에 PASS했던 코드를 그대로 사용합니다.

```scala
package myvexsoc

import spinal.core._
import spinal.lib._
import vexriscv.plugin._

class SimpleDataRam extends Component {

  val io = slave(DBusSimpleBus())

  // 256 words x 32 bits = 1 KiB
  val mem = Mem(Bits(32 bits), wordCount = 256)

  // Byte address -> word address
  val wordAddress = io.cmd.address(9 downto 2)

  // Load request
  val readEnable = io.cmd.fire && !io.cmd.wr

  // Synchronous read
  val readData = mem.readSync(
    address = wordAddress,
    enable = readEnable
  )

  // Store
  mem.write(
    address = wordAddress,
    data = io.cmd.payload.data,
    enable = io.cmd.fire && io.cmd.wr,
    mask = io.cmd.mask
  )

  // Accept every request
  io.cmd.ready := True

  // Load response
  io.rsp.ready := RegNext(readEnable) init (False)
  io.rsp.data := readData
  io.rsp.error := False
}
```

# 5. `VexRiscvTop.scala`

```scala
package myvexsoc

import spinal.core._
import spinal.core.sim._
import vexriscv._
import vexriscv.plugin._

class VexRiscvTop extends Component {

  val cpuConfig = CpuConfig.smallest()

  val system = new ClockingArea(ClockDomain.current) {

    val cpu = new VexRiscv(cpuConfig)

    val instructionRom = new InstructionRom
    val dataRam = new SimpleDataRam
    val dbusDecoder = new DBusDecoder
    val gpio = new SimpleGpio

    // ----------------------------------------------------------
    // IBus simulation visibility
    // ----------------------------------------------------------

    instructionRom.io.cmd.valid.simPublic()
    instructionRom.io.cmd.ready.simPublic()
    instructionRom.io.cmd.pc.simPublic()

    instructionRom.io.rsp.valid.simPublic()
    instructionRom.io.rsp.inst.simPublic()
    instructionRom.io.rsp.error.simPublic()

    // ----------------------------------------------------------
    // Interrupts
    // ----------------------------------------------------------

    val externalInterrupt = False
    val timerInterrupt = False

    // ----------------------------------------------------------
    // Plugin connections
    // ----------------------------------------------------------

    for (plugin <- cpu.plugins) {

      plugin match {

        case p: IBusSimplePlugin =>

          instructionRom.io.cmd <> p.iBus.cmd
          p.iBus.rsp <> instructionRom.io.rsp

        case p: DBusSimplePlugin =>

          // CPU DBus observation
          p.dBus.cmd.valid.simPublic()
          p.dBus.cmd.ready.simPublic()
          p.dBus.cmd.wr.simPublic()
          p.dBus.cmd.address.simPublic()
          p.dBus.cmd.payload.data.simPublic()
          p.dBus.cmd.mask.simPublic()

          p.dBus.rsp.ready.simPublic()
          p.dBus.rsp.data.simPublic()
          p.dBus.rsp.error.simPublic()

          // GPIO observation
          gpio.io.value.simPublic()

          // CPU -> Decoder
          dbusDecoder.io.cpu <> p.dBus

          // Decoder -> peripherals
          dataRam.io <> dbusDecoder.io.ram
          gpio.io.bus <> dbusDecoder.io.gpio

        case p: CsrPlugin =>

          p.externalInterrupt := externalInterrupt
          p.timerInterrupt := timerInterrupt

        case _ =>
      }
    }
  }
}

// ============================================================
// Simulation
// ============================================================

object VexRiscvTopSim extends App {

  import scala.collection.mutable
  import vexriscv.plugin._

  SimConfig.withWave
    .compile(new VexRiscvTop)
    .doSim { dut =>

      println("========================================")
      println(" VexRiscv DBus Decoder + GPIO Test")
      println("========================================")

      dut.clockDomain.forkStimulus(period = 10)
      dut.clockDomain.waitSampling(5)

      println("[INFO] Reset/clock startup complete")

      val dbusPlugin = dut.system.cpu.plugins
        .collectFirst {
          case p: DBusSimplePlugin => p
        }
        .getOrElse(
          throw new Exception("DBusSimplePlugin not found")
        )

      val dbus = dbusPlugin.dBus

      val stores = mutable.Map.empty[Long, Long]

      var storeCount = 0
      var loadCount = 0
      var loadResponseCount = 0
      var loadPending = false

      // --------------------------------------------------------
      // Observe DBus for 100 cycles
      // --------------------------------------------------------

      for (cycle <- 1 to 100) {

        dut.clockDomain.waitFallingEdge()

        // ------------------------------------------------------
        // 1. Observe Load response first
        // ------------------------------------------------------

        val rspReady = dbus.rsp.ready.toBoolean

        if (rspReady) {

          assert(
            loadPending,
            s"Unexpected Load response at cycle $cycle"
          )

          val rspData = dbus.rsp.data.toLong

          println(
            f"Cycle $cycle%3d : LOAD RSP data=0x$rspData%08X"
          )

          assert(
            rspData == 0x00000001L,
            f"Unexpected Load data: 0x$rspData%08X"
          )

          loadResponseCount += 1
          loadPending = false
        }

        // ------------------------------------------------------
        // 2. Observe DBus command
        // ------------------------------------------------------

        val cmdValid = dbus.cmd.valid.toBoolean
        val cmdReady = dbus.cmd.ready.toBoolean

        if (cmdValid && cmdReady) {

          val wr = dbus.cmd.wr.toBoolean
          val address = dbus.cmd.address.toLong
          val data = dbus.cmd.payload.data.toLong
          val mask = dbus.cmd.mask.toInt

          if (wr) {

            storeCount += 1

            stores(address) = data

            println(
              f"Cycle $cycle%3d : STORE addr=0x$address%08X " +
                f"data=0x$data%08X mask=0x$mask%X"
            )

            if (address == 0x10000000L) {
              assert(
                data == 0x00000055L,
                f"Unexpected GPIO data: 0x$data%08X"
              )
            } else {
              assert(
                mask == 0xF,
                f"Unexpected RAM Store mask: 0x$mask%X"
              )
            }

          } else {

            loadCount += 1
            loadPending = true

            println(
              f"Cycle $cycle%3d : LOAD  addr=0x$address%08X"
            )

            assert(
              address == 0x00000000L,
              f"Unexpected Load address: 0x$address%08X"
            )
          }
        }
      }

      // --------------------------------------------------------
      // Result
      // --------------------------------------------------------

      println()
      println("========================================")
      println(" DBus Decoder + GPIO Test Result")
      println("========================================")

      val gpioValue = dut.system.gpio.io.value.toLong

      assert(
        storeCount == 4,
        s"Expected 4 Stores, observed $storeCount"
      )

      assert(
        loadCount == 1,
        s"Expected 1 Load, observed $loadCount"
      )

      assert(
        loadResponseCount == 1,
        s"Expected 1 Load response, observed $loadResponseCount"
      )

      assert(
        !loadPending,
        "Load response is still pending"
      )

      assert(
        stores.getOrElse(0x00L, -1L) == 0x00000001L,
        "RAM[0] Store verification failed"
      )

      assert(
        stores.getOrElse(0x04L, -1L) == 0x00000002L,
        "RAM[1] Store verification failed"
      )

      assert(
        stores.getOrElse(0x08L, -1L) == 0x00000001L,
        "RAM[2] Store verification failed"
      )

      assert(
        stores.getOrElse(0x10000000L, -1L) == 0x00000055L,
        "GPIO Store verification failed"
      )

      assert(
        gpioValue == 0x00000055L,
        f"GPIO value mismatch: 0x$gpioValue%08X"
      )

      println("[PASS] RAM[0] = 0x00000001")
      println("[PASS] RAM[1] = 0x00000002")
      println("[PASS] RAM[2] = 0x00000001")
      println("[PASS] Load response = 0x00000001")
      println("[PASS] GPIO value = 0x00000055")
      println("[PASS] Store count = 4")
      println("[PASS] Load count = 1")

      println("========================================")
      println(" DBus Decoder + GPIO Test finished")
      println("========================================")
    }
}
```

## `VexRiscvTopSim.scala`

```scala id="58321" title="VexRiscvTopSim.scala"
package myvexsoc

import spinal.core.sim._
import vexriscv.plugin._

import scala.collection.mutable

object VexRiscvTopSim extends App {

  SimConfig.withWave
    .compile(new VexRiscvTop)
    .doSim { dut =>

      println("========================================")
      println(" VexRiscv DBus Decoder + GPIO Test")
      println("========================================")

      dut.clockDomain.forkStimulus(period = 10)

      // CPU가 실행을 시작할 수 있도록 초기 clock 진행
      dut.clockDomain.waitSampling(5)

      println("[INFO] Reset/clock startup complete")

      // --------------------------------------------------------
      // DBus Plugin 찾기
      // --------------------------------------------------------

      val dbusPlugin = dut.system.cpu.plugins
        .collectFirst {
          case p: DBusSimplePlugin => p
        }
        .getOrElse {
          throw new Exception("DBusSimplePlugin not found")
        }

      val dbus = dbusPlugin.dBus

      // --------------------------------------------------------
      // Test bookkeeping
      // --------------------------------------------------------

      val stores = mutable.Map.empty[Long, Long]

      var storeCount = 0
      var loadCount = 0
      var loadResponseCount = 0

      // CPU가 Load를 발생시켰고 아직 response를 받지 못한 상태
      var loadPending = false

      // --------------------------------------------------------
      // Observe CPU DBus
      // --------------------------------------------------------

      for (cycle <- 1 to 100) {

        /*
         * falling edge에서 관찰한다.
         *
         * 현재 cycle의 valid/ready를 보고
         * 다음 rising edge에서 실제 command가 fire되는 구조다.
         *
         * Load response는 command 관찰보다 먼저 확인한다.
         */
        dut.clockDomain.waitFallingEdge()

        // ======================================================
        // 1. Load response 관찰
        // ======================================================

        val rspValid = dbus.rsp.ready.toBoolean

        if (rspValid) {

          assert(
            loadPending,
            s"Unexpected Load response at cycle $cycle"
          )

          val rspData = dbus.rsp.data.toLong

          println(
            f"Cycle $cycle%3d : LOAD RSP data=0x$rspData%08X"
          )

          // 이번 프로그램의 유일한 Load:
          //
          // LW x3, 0(x0)
          //
          // RAM[0] = 1 이므로 response는 1이어야 한다.
          assert(
            rspData == 0x00000001L,
            f"Unexpected Load data: 0x$rspData%08X"
          )

          loadResponseCount += 1
          loadPending = false
        }

        // ======================================================
        // 2. DBus command 관찰
        // ======================================================

        val cmdValid = dbus.cmd.valid.toBoolean
        val cmdReady = dbus.cmd.ready.toBoolean

        if (cmdValid && cmdReady) {

          val wr = dbus.cmd.wr.toBoolean
          val address = dbus.cmd.address.toLong
          val data = dbus.cmd.payload.data.toLong
          val mask = dbus.cmd.mask.toInt

          if (wr) {

            // --------------------------------------------------
            // Store
            // --------------------------------------------------

            storeCount += 1
            stores(address) = data

            println(
              f"Cycle $cycle%3d : STORE " +
                f"addr=0x$address%08X " +
                f"data=0x$data%08X " +
                f"mask=0x$mask%X"
            )

            if (address == 0x10000000L) {

              // GPIO Store
              assert(
                data == 0x00000055L,
                f"Unexpected GPIO data: 0x$data%08X"
              )

            } else {

              // RAM Store
              assert(
                address == 0x00000000L ||
                  address == 0x00000004L ||
                  address == 0x00000008L,
                f"Unexpected RAM Store address: 0x$address%08X"
              )

              assert(
                mask == 0xF,
                f"Unexpected RAM Store mask: 0x$mask%X"
              )
            }

          } else {

            // --------------------------------------------------
            // Load
            // --------------------------------------------------

            loadCount += 1
            loadPending = true

            println(
              f"Cycle $cycle%3d : LOAD  addr=0x$address%08X"
            )

            // 이번 테스트에서는
            //
            // LW x3, 0(x0)
            //
            // 하나만 수행한다.
            assert(
              address == 0x00000000L,
              f"Unexpected Load address: 0x$address%08X"
            )
          }
        }
      }

      // ========================================================
      // Final result
      // ========================================================

      println()
      println("========================================")
      println(" DBus Decoder + GPIO Test Result")
      println("========================================")

      // GPIO register 최종값
      val gpioValue = dut.system.gpio.io.value.toLong

      // --------------------------------------------------------
      // Transaction count
      // --------------------------------------------------------

      assert(
        storeCount == 4,
        s"Expected 4 Stores, observed $storeCount"
      )

      assert(
        loadCount == 1,
        s"Expected 1 Load, observed $loadCount"
      )

      assert(
        loadResponseCount == 1,
        s"Expected 1 Load response, observed $loadResponseCount"
      )

      assert(
        !loadPending,
        "Load response is still pending"
      )

      // --------------------------------------------------------
      // RAM Store verification
      // --------------------------------------------------------

      assert(
        stores.getOrElse(0x00000000L, -1L) == 0x00000001L,
        "RAM[0] Store verification failed"
      )

      assert(
        stores.getOrElse(0x00000004L, -1L) == 0x00000002L,
        "RAM[1] Store verification failed"
      )

      assert(
        stores.getOrElse(0x00000008L, -1L) == 0x00000001L,
        "RAM[2] Store verification failed"
      )

      // --------------------------------------------------------
      // GPIO Store verification
      // --------------------------------------------------------

      assert(
        stores.getOrElse(0x10000000L, -1L) == 0x00000055L,
        "GPIO Store verification failed"
      )

      assert(
        gpioValue == 0x00000055L,
        f"GPIO value mismatch: 0x$gpioValue%08X"
      )

      // ========================================================
      // PASS messages
      // ========================================================

      println("[PASS] RAM[0] = 0x00000001")
      println("[PASS] RAM[1] = 0x00000002")
      println("[PASS] RAM[2] = 0x00000001")
      println("[PASS] Load response = 0x00000001")
      println("[PASS] GPIO value = 0x00000055")
      println("[PASS] Store count = 4")
      println("[PASS] Load count = 1")

      println("========================================")
      println(" DBus Decoder + GPIO Test finished")
      println("========================================")
    }
}
```

### 이 테스트가 확인하는 것

이번 테스트 프로그램의 실제 DBus 동작은:

```text
RAM
0x00000000 ← 1
0x00000004 ← 2

GPIO
0x10000000 ← 0x55

RAM
0x00000000 → 1
0x00000008 ← 1
```

입니다.

특히 마지막의

```text
RAM[0] → LW → x3 → SW → RAM[2]
```

를 확인하므로 **RAM의 Load response가 Decoder를 거쳐 CPU까지 정상적으로 돌아왔는지**도 검증합니다.

그리고

```text
GPIO Store → DBusDecoder → SimpleGpio.gpioValue
```

도 별도로 확인합니다.

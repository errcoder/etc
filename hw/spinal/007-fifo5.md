# 좋다. 이제 FIFO **Pointer Wrap-Around**를 직접 검증하자

이번 단계의 목표는 단순하다.

> `writePtr`와 `readPtr`가 FIFO 끝에 도달하면 `0`으로 정상적으로 돌아가는지 확인한다.

그리고 이번에는 **RTL 코드 + Simulation 코드 + 예상 결과 + GTKWave에서 볼 포인트**까지 한 번에 가자.

---

# 1. `SimpleFifo`는 그대로 두고 `simPublic()`만 추가

SpinalHDL **Release 34 / v1.15.0** 기준이다.

```scala
import spinal.core._
import spinal.core.sim._

case class SimpleFifo(
    dataWidth: Int,
    depth: Int
) extends Component {

  require(depth >= 2)
  require(dataWidth >= 1)

  val io = new Bundle {
    val write     = in Bool()
    val writeData = in Bits(dataWidth bits)

    val read      = in Bool()
    val readData  = out Bits(dataWidth bits)

    val full      = out Bool()
    val empty     = out Bool()
  }

  // -------------------------
  // Memory
  // -------------------------

  val mem = Vec(
    Reg(Bits(dataWidth bits)) init(0),
    depth
  )

  // -------------------------
  // Pointer
  // -------------------------

  val ptrWidth =
    log2Up(depth)

  val writePtr =
    Reg(UInt(ptrWidth bits)) init(0)

  val readPtr =
    Reg(UInt(ptrWidth bits)) init(0)

  // Simulation에서 내부 상태 확인
  writePtr.simPublic()
  readPtr.simPublic()

  // -------------------------
  // Count
  // -------------------------

  val countWidth =
    log2Up(depth + 1)

  val count =
    Reg(UInt(countWidth bits)) init(0)

  count.simPublic()

  // -------------------------
  // Status
  // -------------------------

  io.empty := count === 0
  io.full  := count === depth

  // FIFO의 현재 head
  io.readData := mem(readPtr)

  // -------------------------
  // Actual operations
  // -------------------------

  val doWrite =
    io.write && !io.full

  val doRead =
    io.read && !io.empty

  // -------------------------
  // Write
  // -------------------------

  when(doWrite) {

    mem(writePtr) := io.writeData

    when(writePtr === depth - 1) {
      writePtr := 0
    } otherwise {
      writePtr := writePtr + 1
    }
  }

  // -------------------------
  // Read
  // -------------------------

  when(doRead) {

    when(readPtr === depth - 1) {
      readPtr := 0
    } otherwise {
      readPtr := readPtr + 1
    }
  }

  // -------------------------
  // Count
  // -------------------------

  when(doWrite && !doRead) {

    count := count + 1

  } elsewhen(!doWrite && doRead) {

    count := count - 1
  }
}
```

# 2. 이번에는 Simulation을 명확하게 작성하자

이번 테스트에서는 다음 순서로 간다.

```text
A write
B write
C write
D write

A read
B read

E write
F write
```

즉:

```text
A B C D
↓ ↓
C D E F
```

가 최종 FIFO 내용이 되어야 한다.

그리고 pointer는 다음처럼 움직여야 한다.

```text
writePtr

0 → 1 → 2 → 3 → 0 → 1 → 2
```

```text
readPtr

0 → 1 → 2
```

---

## Simulation 코드

```scala
import spinal.core._
import spinal.core.sim._

object SimpleFifoSim extends App {

  val depth = 4

  SimConfig
    .withWave
    .doSim(SimpleFifo(
      dataWidth = 8,
      depth = depth
    )) { dut =>

      dut.clockDomain.forkStimulus(10)

      // -------------------------
      // Helper
      // -------------------------

      def printStatus(label: String): Unit = {
        println(
          f"$label%-18s " +
          f"writePtr=${dut.writePtr.toInt}%d " +
          f"readPtr=${dut.readPtr.toInt}%d " +
          f"count=${dut.count.toInt}%d " +
          f"full=${dut.io.full.toBoolean} " +
          f"empty=${dut.io.empty.toBoolean} " +
          f"readData=0x${dut.io.readData.toBigInt}%02X"
        )
      }

      def writeOne(data: Int): Unit = {
        dut.io.writeData #= data
        dut.io.write #= true

        dut.clockDomain.waitSampling()

        dut.io.write #= false
      }

      def readOne(): Unit = {
        dut.io.read #= true

        dut.clockDomain.waitSampling()

        dut.io.read #= false
      }

      // -------------------------
      // Initial
      // -------------------------

      dut.io.write     #= false
      dut.io.writeData #= 0
      dut.io.read      #= false

      dut.clockDomain.waitSampling(2)

      printStatus("Initial")

      assert(dut.io.empty.toBoolean)
      assert(!dut.io.full.toBoolean)

      // -------------------------
      // Write A
      // -------------------------

      writeOne(0x41)

      assert(dut.io.readData.toBigInt == 0x41)
      printStatus("After write A")

      // -------------------------
      // Write B
      // -------------------------

      writeOne(0x42)

      printStatus("After write B")

      // -------------------------
      // Write C
      // -------------------------

      writeOne(0x43)

      printStatus("After write C")

      // -------------------------
      // Write D
      // -------------------------

      writeOne(0x44)

      printStatus("After write D")

      assert(dut.io.full.toBoolean)
      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 4)
      assert(dut.io.readData.toBigInt == 0x41)

      // -------------------------
      // Read A
      // -------------------------

      readOne()

      dut.clockDomain.waitFallingEdge()

      printStatus("After read A")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 1)
      assert(dut.count.toInt == 3)
      assert(dut.io.readData.toBigInt == 0x42)

      // -------------------------
      // Read B
      // -------------------------

      readOne()

      dut.clockDomain.waitFallingEdge()

      printStatus("After read B")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 2)
      assert(dut.count.toInt == 2)
      assert(dut.io.readData.toBigInt == 0x43)

      // -------------------------
      // Write E
      // -------------------------

      writeOne(0x45)

      printStatus("After write E")

      assert(dut.writePtr.toInt == 1)
      assert(dut.readPtr.toInt == 2)
      assert(dut.count.toInt == 3)
      assert(dut.io.readData.toBigInt == 0x43)

      // -------------------------
      // Write F
      // -------------------------

      writeOne(0x46)

      printStatus("After write F")

      assert(dut.writePtr.toInt == 2)
      assert(dut.readPtr.toInt == 2)
      assert(dut.count.toInt == 4)
      assert(dut.io.full.toBoolean)
      assert(dut.io.readData.toBigInt == 0x43)

      println("[Done] FIFO wrap-around test passed.")
    }
}
```

# 3. 그런데 여기서 한 가지 주의

이번 코드에서:

```scala
def writeOne(data: Int): Unit = {
  dut.io.writeData #= data
  dut.io.write #= true

  dut.clockDomain.waitSampling()

  dut.io.write #= false
```

뒤에 바로 다음 `writeOne()`이 이어진다.

즉:

```text

```

사이에서 `write=false` 상태가 **별도의 sampling event를 거치지 않는다.**

그래서 실제 DUT 입장에서는:

```text
A write
↓
sampling
↓
B write
↓
sampling
↓
C write
```

가 된다.

이 부분은 우리가 앞에서 확인했던 **SpinalSim scheduling 문제**와 연결된다.

이번 테스트에서는 이 동작을 의도적으로 이용하고 있다.

# 5. 여기서 정말 중요한 부분

## D를 쓴 순간

```text
writePtr = 0
```

이다.

왜?

D는 `mem[3]`에 저장되기 때문이다.

```text
writePtr = 3
```

상태에서 write가 실행되고:

```scala
when(writePtr === depth - 1) {
  writePtr := 0
}
```

가 실행된다.

즉:

```text
D
↓
mem[3]에 저장
↓
writePtr = 0
```

이다.

**다음 write는 다시 `mem[0]`부터 시작한다.**

---

# 6. E를 쓰는 순간

이미 A를 읽었기 때문에:

```text
mem[0] = A
```

는 더 이상 FIFO의 유효한 데이터가 아니다.

현재:

```text
writePtr = 0
readPtr  = 2
count    = 2
```

이므로 E를 쓰면:

```text
mem[0] = E
```

그리고:

```text
writePtr = 1
count = 3
```

가 된다.

FIFO의 논리적인 내용은:

```text
C D E
```

이다.

메모리 물리 위치는:

```text
index 0 = E
index 1 = B
index 2 = C
index 3 = D
```

이다.

여기서 **논리적인 FIFO 순서와 메모리 배열의 물리적인 순서는 다르다**는 중요한 개념이 나온다.

---

# 7. F까지 쓰면

F는:

```text
mem[1] = F
```

에 저장된다.

그러면 실제 메모리는:

```text
index 0    E
index 1    F
index 2    C
index 3    D
```

하지만 FIFO의 순서는:

```text
C → D → E → F
```

이다.

왜냐하면 `readPtr`가 2이기 때문이다.

```text
readPtr = 2
```

에서 시작해서:

```text
2 → 3 → 0 → 1
```

순서로 읽기 때문이다.

이게 **Circular FIFO**의 핵심이다.

# 10. 그리고 다음 단계가 아주 중요하다

이번 실험까지 통과하면 FIFO의 기본 구조는 상당히 확실해진다.

그 다음에는 **경계 조건(boundary condition)**을 테스트해야 한다.

순서는 이렇게 가는 게 좋다.

```text
① Pointer wrap-around       ← 지금
        ↓
② EMPTY + READ
        ↓
③ FULL + WRITE
        ↓
④ FULL + READ + WRITE
        ↓
⑤ EMPTY + READ + WRITE
        ↓
⑥ 동시에 READ + WRITE
        ↓
⑦ FIFO 정책 개선
        ↓
⑧ Vec(Reg(...)) → Mem(...)
```

특히 **② EMPTY + READ**와 **③ FULL + WRITE**는 아주 중요하다.

왜냐하면 현재 RTL의 핵심 정책이 바로 이것이기 때문이다.

```scala
val doWrite =
  io.write && !io.full

val doRead =
  io.read && !io.empty
```

따라서 다음 단계에서는 **"읽으라고 했는데 FIFO가 비어 있으면 실제로 아무 일도 일어나지 않는가?"**, 그리고 **"가득 찼는데 쓰려고 하면 정말 상태가 그대로 유지되는가?"**를 waveform으로 검증해보자.

# 3. 왜 D를 썼는데 `writePtr=0`인가?

D를 쓰기 **직전**에는:

```text
writePtr = 3
```

이다.

그리고 FIFO depth가 4이므로 마지막 위치는:

```text
depth - 1 = 3
```

이다.

RTL:

```scala
when(doWrite) {

  mem(writePtr) := io.writeData

  when(writePtr === depth - 1) {
    writePtr := 0
  } otherwise {
    writePtr := writePtr + 1
  }
}
```

D가 들어오면:

```text
mem[3] = D
```

그리고:

```text
writePtr = 0
```

이 된다.

즉:

```text
writePtr

0 → 1 → 2 → 3 → 0
                  ↑
               wrap-around
```

첫 번째 **circular pointer**가 확인되는 것이다.

# 7. 여기서 FIFO의 핵심 구조가 보인다

지금까지의 상태를 그림으로 보면:

```text
                    writePtr
                       ↓
                 ┌───────┐
                 │       │
                 ▼       │
       ┌─────┬─────┬─────┬─────┐
index  │  0  │  1  │  2  │  3  │
       ├─────┼─────┼─────┼─────┤
       │  E  │  F  │  C  │  D  │
       └─────┴─────┴─────┴─────┘
                    ↑
                 readPtr
```

물리적인 memory 배열은:

```text
E F C D
```

인데,

FIFO가 바라보는 논리적인 데이터 순서는:

```text
C → D → E → F
```

이다.

**FIFO 순서를 결정하는 것은 memory 배열의 앞에서부터가 아니라 `readPtr`와 `writePtr`다.**

# 8. 이제 다음 실험으로 넘어갈 준비가 됐다

여기까지 통과하면 다음은 아주 중요한 **boundary condition**이다.

### EMPTY + READ

현재 FIFO가:

```text
count = 0
empty = true
```

인데:

```text
read = 1
```

을 넣으면?

우리가 작성한 RTL에서는:

```scala
val doRead =
  io.read && !io.empty
```

이므로:

```text
read     = 1
empty    = 1
----------------
doRead   = 0
```

이어야 한다.

따라서:

```text
readPtr
writePtr
count
```

가 **전혀 변하지 않아야 한다.**

---

그리고 반대쪽의:

### FULL + WRITE

도 확인해야 한다.

```text
count = depth
full  = true
```

에서:

```text
write = 1
```

이면:

```scala
val doWrite =
  io.write && !io.full
```

이므로:

```text
doWrite = false
```

다.

따라서 이것 역시 FIFO 상태가 변하지 않아야 한다.

---

**즉 다음 단계에서는 `EMPTY + READ`부터 검증하자.**

이번에는 특히 우리가 이미 한 번 simulation scheduling 문제를 겪었으므로, **상태를 바꾸는 clock edge와 상태를 검사하는 시점을 분리한 형태로 테스트**하겠다.

# 좋다. 이제 `EMPTY + READ` 경계 조건을 검증하자

이번 테스트의 질문은 아주 명확하다.

> **FIFO가 비어 있을 때 `read=1`을 넣어도 FIFO 내부 상태가 절대로 변하지 않는가?**

현재 RTL의 핵심은:

```scala
val doRead =
  io.read && !io.empty
```

이므로 FIFO가 비어 있으면:

```text
io.read  = true
io.empty = true

doRead = true && false
       = false
```

즉 **READ 요청은 무시되어야 한다.**

---

# 1. 테스트 코드는 이렇게 작성하자

이번에는 처음부터 EMPTY 상태에서 시작해서 READ를 걸어본다.

```scala
import spinal.core._
import spinal.core.sim._

object SimpleFifoEmptyReadSim extends App {

  SimConfig
    .withWave
    .doSim(SimpleFifo(
      dataWidth = 8,
      depth = 4
    )) { dut =>

      dut.clockDomain.forkStimulus(10)

      def printStatus(label: String): Unit = {
        println(
          f"$label%-22s " +
          f"writePtr=${dut.writePtr.toInt}%d " +
          f"readPtr=${dut.readPtr.toInt}%d " +
          f"count=${dut.count.toInt}%d " +
          f"full=${dut.io.full.toBoolean} " +
          f"empty=${dut.io.empty.toBoolean} " +
          f"readData=0x${dut.io.readData.toBigInt}%02X"
        )
      }

      // -------------------------
      // Initial
      // -------------------------

      dut.io.write     #= false
      dut.io.writeData #= 0
      dut.io.read      #= false

      dut.clockDomain.waitSampling(2)

      printStatus("Initial")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 0)

      assert(dut.io.empty.toBoolean)
      assert(!dut.io.full.toBoolean)

      // -------------------------
      // EMPTY + READ
      // -------------------------

      dut.io.read #= true

      dut.clockDomain.waitSampling()

      // READ 요청을 제거
      dut.io.read #= false

      // 상태가 반영된 뒤 관찰
      dut.clockDomain.waitFallingEdge()

      printStatus("After empty read")

      // -------------------------
      // Verify
      // -------------------------

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 0)

      assert(dut.io.empty.toBoolean)
      assert(!dut.io.full.toBoolean)

      println("[Done] EMPTY + READ test passed.")
    }
}
```

# 2. 예상 결과

정상이면:

```text
Initial                writePtr=0 readPtr=0 count=0 full=false empty=true readData=0x00

After empty read       writePtr=0 readPtr=0 count=0 full=false empty=true readData=0x00

[Done] EMPTY + READ test passed.
```

핵심은 **모든 내부 상태가 그대로**라는 것이다.

```text
writePtr : 0 → 0
readPtr  : 0 → 0
count    : 0 → 0
```

# 3. 왜 아무것도 변하지 않는가?

RTL을 다시 보자.

```scala
val doRead =
  io.read && !io.empty
```

EMPTY 상태에서는:

```text
io.read  = 1
io.empty = 1
```

이므로:

```text
doRead = 1 && !1
       = 1 && 0
       = 0
```

따라서 이 코드가 실행되지 않는다.

```scala
when(doRead) {

  when(readPtr === depth - 1) {
    readPtr := 0
  } otherwise {
    readPtr := readPtr + 1
  }
}
```

그리고 count도:

```scala
when(doWrite && !doRead) {
  count := count + 1
} elsewhen(!doWrite && doRead) {
  count := count - 1
}
```

여기서:

```text
doWrite = 0
doRead  = 0
```

이므로 어느 조건도 참이 아니다.

따라서:

```text
count = 0
```

그대로 유지된다.

---

# 4. 그런데 `readData`는 왜 0인가?

현재 FIFO는 초기화할 때:

```scala
val mem = Vec(
  Reg(Bits(dataWidth bits)) init(0),
  depth
)
```

이렇게 되어 있다.

그래서:

```text
mem[0] = 0
mem[1] = 0
mem[2] = 0
mem[3] = 0
```

이고:

```text
readPtr = 0
```

이므로:

```scala
io.readData := mem(readPtr)
```

결과는:

```text
readData = mem[0] = 0x00
```

이다.

중요한 점은 **EMPTY 상태에서 `readData`가 0이어야 한다는 것이 FIFO의 필수 조건은 아니다.**

현재 구현에서는 초기화된 memory 때문에 0으로 보이는 것이다.

FIFO를 사용하는 쪽에서는 일반적으로:

```text
empty = true
```

이면 `readData`를 유효한 데이터로 사용하면 안 된다.

즉:

```text
empty = true
readData = 0x00
```

이든

```text
empty = true
readData = 0x43
```

이든 **현재 FIFO 인터페이스의 의미상 큰 차이가 없다.**

유효성을 결정하는 것은:

```text
empty
```

이다.

---

# 5. 아주 중요한 개념

FIFO의 `read`는 단순히:

> "readData를 보여줘."

라는 의미가 아니다.

현재 구조에서는:

```text
read = 1
```

이면

> **"현재 head를 소비하고 다음 데이터로 넘어가라."**

라는 의미다.

그래서 EMPTY 상태에서는 소비할 데이터가 없으므로:

```text
read = 1
empty = 1
       ↓
아무 동작도 하지 않음
```

이 된다.

# 7. 그리고 그 다음은 FULL + WRITE

FIFO를:

```text
A B C D
```

로 가득 채운 다음:

```text
write = 1
writeData = E
```

를 넣는다.

현재 정책에서는:

```scala
val doWrite =
  io.write && !io.full
```

이므로:

```text
full = true
write = true
       ↓
doWrite = false
```

가 된다.

따라서:

```text
writePtr
readPtr
count
readData
```

모두 변하지 않아야 한다.

즉:

```text
A B C D
```

가 그대로 유지되고 E는 들어가지 않는다.

# 좋습니다. EMPTY + READ 테스트 통과했습니다.

## 이번 테스트에서 확인한 것

FIFO가 비어 있을 때:

```scala
io.read = true
```

를 줘도 실제 읽기는 수행되지 않습니다.

현재 RTL의 핵심은:

```scala
val doRead =

```

```text
empty = true

```

이면

```text
doRead = true && false
       = false
```

입니다.

따라서:

| 항목 | 읽기 전 | EMPTY + READ 후 |
|---|---:|---:|
| writePtr | 0 | 0 |
| readPtr | 0 | 0 |
| count | 0 | 0 |
| empty | true | true |
| full | false | false |

즉 **FIFO 상태가 전혀 변하지 않았습니다.**

# 다음 테스트: FULL + WRITE

이제 반대쪽 경계 조건을 확인해봅시다.

FIFO를 먼저:

```text
A
B
C
D
```

로 가득 채운 다음,

```text
E
```

를 추가로 써보겠습니다.

현재 FIFO 정책에서는:

```scala
val doWrite =
  io.write && !io.full
```

이므로 FULL 상태에서 WRITE를 요청하면:

```text
full  = true
write = true

doWrite = true && false
       = false
```

따라서 **E는 버려지고 FIFO 상태는 변하지 않아야 합니다.**

## 테스트 코드

```scala
import spinal.core._
import spinal.core.sim._

object SimpleFifoFullWriteSim extends App {

  SimConfig
    .withWave
    .doSim(SimpleFifo(
      dataWidth = 8,
      depth = 4
    )) { dut =>

      dut.clockDomain.forkStimulus(10)

      def printStatus(label: String): Unit = {
        println(
          f"$label%-22s " +
          f"writePtr=${dut.writePtr.toInt}%d " +
          f"readPtr=${dut.readPtr.toInt}%d " +
          f"count=${dut.count.toInt}%d " +
          f"full=${dut.io.full.toBoolean} " +
          f"empty=${dut.io.empty.toBoolean} " +
          f"readData=0x${dut.io.readData.toBigInt}%02X"
        )
      }

      def writeOne(data: Int): Unit = {
        dut.io.writeData #= data
        dut.io.write #= true

        dut.clockDomain.waitSampling()

        dut.io.write #= false
        dut.clockDomain.waitFallingEdge()
      }

      // -------------------------
      // Initial
      // -------------------------

      dut.io.write     #= false
      dut.io.writeData #= 0
      dut.io.read      #= false

      dut.clockDomain.waitSampling(2)

      printStatus("Initial")

      // -------------------------
      // Fill FIFO
      // A, B, C, D
      // -------------------------

      writeOne(0x41)
      writeOne(0x42)
      writeOne(0x43)
      writeOne(0x44)

      printStatus("FIFO full")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 4)

      assert(dut.io.full.toBoolean)
      assert(!dut.io.empty.toBoolean)

      assert(dut.io.readData.toBigInt == 0x41)

      // -------------------------
      // FULL + WRITE
      // Try to write E
      // -------------------------

      dut.io.writeData #= 0x45
      dut.io.write #= true

      dut.clockDomain.waitSampling()

      dut.io.write #= false
      dut.clockDomain.waitFallingEdge()

      printStatus("After full write")

      // -------------------------
      // State must NOT change
      // -------------------------

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 4)

      assert(dut.io.full.toBoolean)
      assert(!dut.io.empty.toBoolean)

      // Head must still be A
      assert(dut.io.readData.toBigInt == 0x41)

      println("[Done] FULL + WRITE test passed.")
    }
}
```


가 되어야 합니다.

좋습니다. 이번에는 **현재 `SimpleFifo` RTL을 그대로 유지**하고, `FULL + READ + WRITE`에서 **현재 정책상 WRITE는 막히고 READ만 수행되는 것**을 검증하겠습니다.

# FULL + READ + WRITE 테스트

```scala id="58321"
import spinal.core._
import spinal.core.sim._

object SimpleFifoFullReadWriteSim extends App {

  SimConfig
    .withWave
    .doSim(SimpleFifo(
      dataWidth = 8,
      depth = 4
    )) { dut =>

      dut.clockDomain.forkStimulus(10)

      def printStatus(label: String): Unit = {
        println(
          f"$label%-24s " +
          f"writePtr=${dut.writePtr.toInt}%d " +
          f"readPtr=${dut.readPtr.toInt}%d " +
          f"count=${dut.count.toInt}%d " +
          f"full=${dut.io.full.toBoolean} " +
          f"empty=${dut.io.empty.toBoolean} " +
          f"readData=0x${dut.io.readData.toBigInt}%02X"
        )
      }

      def writeOne(data: Int): Unit = {
        dut.io.writeData #= data
        dut.io.write #= true

        dut.clockDomain.waitSampling()

        dut.io.write #= false
        dut.clockDomain.waitFallingEdge()
      }

      // --------------------------------------------------
      // Initial
      // --------------------------------------------------

      dut.io.write     #= false
      dut.io.writeData #= 0
      dut.io.read      #= false

      dut.clockDomain.waitSampling(2)

      printStatus("Initial")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 0)

      assert(dut.io.empty.toBoolean)
      assert(!dut.io.full.toBoolean)

      // --------------------------------------------------
      // Fill FIFO
      //
      // FIFO = A B C D
      //        ^
      //       head
      // --------------------------------------------------

      writeOne(0x41) // A
      writeOne(0x42) // B
      writeOne(0x43) // C
      writeOne(0x44) // D

      printStatus("FIFO full")

      assert(dut.writePtr.toInt == 0)
      assert(dut.readPtr.toInt == 0)
      assert(dut.count.toInt == 4)

      assert(dut.io.full.toBoolean)
      assert(!dut.io.empty.toBoolean)

      assert(dut.io.readData.toBigInt == 0x41)

      // --------------------------------------------------
      // FULL + READ + WRITE
      //
      // Try:
      //   READ  A
      //   WRITE E
      //
      // Current FIFO policy:
      //
      //   doWrite = write && !full
      //   doRead  = read  && !empty
      //
      // Therefore:
      //
      //   doWrite = false
      //   doRead  = true
      //
      // Result:
      //   A is removed
      //   E is NOT written
      //
      // FIFO becomes:
      //
      //   B C D
      //   ^
      //  head
      // --------------------------------------------------

      dut.io.writeData #= 0x45 // E
      dut.io.write     #= true
      dut.io.read      #= true

      dut.clockDomain.waitSampling()

      dut.io.write #= false
      dut.io.read  #= false

      dut.clockDomain.waitFallingEdge()

      printStatus("After full read+write")

      // --------------------------------------------------
      // Verify READ happened
      // --------------------------------------------------

      assert(
        dut.readPtr.toInt == 1,
        "readPtr should move from 0 to 1"
      )

      assert(
        dut.count.toInt == 3,
        "count should decrease from 4 to 3"
      )

      // --------------------------------------------------
      // Verify WRITE did NOT happen
      // --------------------------------------------------

      assert(
        dut.writePtr.toInt == 0,
        "writePtr should remain 0 because WRITE is blocked while full"
      )

      assert(
        dut.io.readData.toBigInt == 0x42,
        "After reading A, FIFO head must be B"
      )

      assert(
        !dut.io.full.toBoolean,
        "FIFO should no longer be full"
      )

      assert(
        !dut.io.empty.toBoolean,
        "FIFO should still contain B, C, D"
      )

      println("[Done] FULL + READ + WRITE test passed.")
    }
}
```
